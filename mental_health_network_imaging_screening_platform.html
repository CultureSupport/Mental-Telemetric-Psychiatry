defmodule SynapseCareWeb.ChatChannel do
  use SynapseCareWeb, :channel
  alias SynapseCareWeb.Presence

  @impl true
  def join("chat:room:" <> room_id, _payload, socket) do
    send(self(), :after_join)
    {:ok, assign(socket, :room_id, room_id)}
  end

  @impl true
  def handle_info(:after_join, socket) do
    {:ok, _} = Presence.track(socket, socket.assigns.user_id, %{
      online_at: inspect(System.system_time(:second)),
      role: socket.assigns.role
    })
    push(socket, "presence_state", Presence.list(socket))
    {:noreply, socket}
  end

  @impl true
  def handle_in("new_msg", %{"body" => body, "encrypted" => true}, socket) do
    msg = %{
      id: Ecto.UUID.generate(),
      user_id: socket.assigns.user_id,
      body: body,
      inserted_at: DateTime.utc_now()
    }
    broadcast!(socket, "new_msg", msg)
    {:reply, {:ok, msg}, socket}
  end

  @impl true
  def handle_in("screening_alert", %{"severity" => "severe", "suicide_flag" => true} = p, socket) do
    # Instant Fan-Out to Clinical Escalation Cluster
    SynapseCareWeb.Endpoint.broadcast("triage:clinicians", "suicide_risk_alert", %{
      patient_id: socket.assigns.user_id,
      score: p["total_score"],
      room_id: socket.assigns.room_id
    })
    {:reply, :ok, socket}
  end
end

(ns synapse.screening-core
  (:require [clojure.spec.alpha :as s]))

(s/def ::answer-val (s/int-in 0 4))
(s/def ::phq9-vector (s/coll-of ::answer-val :count 9))

(def phq9-thresholds
  [{:max 4  :label "Minimal"           :action "Routine Follow-up"}
   {:max 9  :label "Mild"              :action "Watchful Waiting"}
   {:max 14 :label "Moderate"          :action "Consider Psychotherapy"}
   {:max 19 :label "Moderately Severe" :action "Active Clinical Intervention"}
   {:max 27 :label "Severe"            :action "Immediate Safety Protocol"}])

(defn evaluate-phq9
  "Pure function evaluating PHQ-9 psychometrics and suicidal ideation."
  [answers]
  {:pre [(s/valid? ::phq9-vector answers)]}
  (let [score (reduce + answers)
        suicide-flag? (pos? (nth answers 8))
        tier (first (filter #(<= score (:max %)) phq9-thresholds))]
    {:total-score score
     :severity    (:label tier)
     :action      (if suicide-flag?
                    "Immediate Safety Escalation"
                    (:action tier))
     :suicide-risk suicide-flag?
     :evaluated-at (System/currentTimeMillis)}))

(defn to-fhir-response
  "Encodes assessment into standard HL7 FHIR R4 QuestionnaireResponse"
  [patient-id assessment-result]
  {:resourceType "QuestionnaireResponse"
   :status "completed"
   :subject {:reference (str "Patient/" patient-id)}
   :authored (java.time.Instant/now)
   :item [{:linkId "total-score" :answer [{:valueInteger (:total-score assessment-result)}]}
          {:linkId "clinical-action" :answer [{:valueString (:action assessment-result)}]}]})
